\subsection{Data and Preprocessing}
We use the Shenzhen taxicab GPS dataset \citep{zhang2015urbancps}. The raw data consists of GPS logs (ID, timestamp, latitude, longitude, occupancy, speed). We preprocess the data as follows:
\begin{enumerate}
    \item \textbf{Grid Rasterization}: The spatial domain is discretized into a $400 \times 800$ grid with coordinates $p_t \in \mathbb{R}^2$ representing $[y, x]$ indices. The bounding box is defined by lat $[22.45, 22.85]$ and lon $[113.75, 114.65]$.
    \item \textbf{Time Re-sampling}: Trajectories are resampled to a fixed interval $\Delta t = 30$s. This enables explicit modeling of velocity as \textit{step displacement}, denoted as $v_t = p_{t+1} - p_t$.
    \item \textbf{Normalization}: We compute the mean $\mu_v$ and standard deviation $\sigma_v$ of velocities from the training split and normalize inputs to zero mean and unit variance.
\end{enumerate}

\subsection{Problem Formulation}
Let $X = \{v_{-H+1}, \dots, v_0\}$ denote the observed velocity sequence of length $H$, and $Y = \{v_1, \dots, v_F\}$ denote the future velocity sequence of length $F$. Given a destination $d$ and context $C$ (including $X$ and $d$), our goal is to model the conditional distribution $p_\theta(Y | C)$.

\subsection{Denoising Diffusion Probabilistic Models (DDPM)}
We formulate the trajectory generation as a reverse diffusion process. The forward process $q(y_1, \dots, y_T | y_0)$ adds Gaussian noise to the clean future trajectory $y_0 = Y$ over $T$ timesteps according to a variance schedule $\beta_1, \dots, \beta_T$:
\begin{equation}
    q(y_t | y_{t-1}) = \mathcal{N}(y_t; \sqrt{1 - \beta_t} y_{t-1}, \beta_t \mathbf{I})
\end{equation}
The reverse process $p_\theta(y_{t-1} | y_t, C)$ denoises the latent variable $y_T \sim \mathcal{N}(0, \mathbf{I})$ to recover $y_0$:
\begin{equation}
    p_\theta(y_{t-1} | y_t, C) = \mathcal{N}(y_{t-1}; \mu_\theta(y_t, t, C), \Sigma_\theta(y_t, t, C))
\end{equation}
We parameterize the model to predict the added noise $\epsilon_\theta(y_t, t, C)$ and optimize the simplified variational lower bound:
\begin{equation}
    \mathcal{L}_{simple} = \mathbb{E}_{y_0, t, \epsilon} \left[ \| \epsilon - \epsilon_\theta(y_t, t, C) \|^2 \right]
\end{equation}

\subsection{Physics-Informed Conditioning: Navigation Field}
To incorporate urban flow dynamics, we construct a \textit{Navigation Field} $\mathcal{F}$ from the training data. For each grid cell $(i, j)$, we estimate the mean velocity vector $\bar{v}_{i,j}$ from all trajectory segments passing through it.
\begin{equation}
    \mathcal{F}(i, j) = \frac{\sum_{k \in \mathcal{K}_{i,j}} v_k}{|\mathcal{K}_{i,j}| + \epsilon}
\end{equation}
where $\mathcal{K}_{i,j}$ is the set of traversing segments. At inference time, we extract a local patch $\Phi(p_t)$ of size $K \times K$ from $\mathcal{F}$ centered at the current position $p_t$, encoding the local flow prior. This patch is fed into a CNN encoder and concatenated with the trajectory embedding, guiding the diffusion process towards physically plausible directions.

\subsection{Training-Time Macro Regularization}
To mitigate the "shrinkage" phenomenon where generated trajectories lack directional persistence, we introduce a macroscopic regularization loss. We approximate the clean trajectory $\hat{y}_0$ from $y_t$ and $\epsilon_\theta(y_t)$ using the DDPM reconstruction formula. We then compute the cumulative displacement (position) from the predicted velocities:
\begin{equation}
    \hat{p}_k = p_0 + \sum_{i=1}^k \hat{v}_i, \quad \hat{v} \in \hat{y}_0
\end{equation}
The macro loss $\mathcal{L}_{macro}$ penalizes the deviation of the end-to-end displacement from the ground truth:
\begin{equation}
    \mathcal{L}_{macro} = \| (\hat{p}_F - p_0) - (p_F - p_0) \|^2
\end{equation}
To prevent the model from optimizing high-frequency noise at large $t$, we gate this loss using a time threshold $\tau_{thr}$, applying it only when $t < \tau_{thr}$.
