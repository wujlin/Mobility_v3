\section{Phase B v1.0: Models \& Diagnosis}

\begin{frame}{Phase B v1.0 (what changes from Phase A?)}
\textbf{Relation to Phase A:} with a strict dt-fixed pipeline, Phase B compares model families under the same protocol.

\vspace{0.6em}
\textbf{Phase B v1.0 question:} does diffusion provide better multi-modal coverage, and does nav\_field improve it?

\vspace{0.6em}
\textbf{Models compared:}
\begin{itemize}
  \item Baseline (deterministic L2 regression, $K{=}1$)
  \item Diffusion (data-only, $K{=}20$)
  \item Physics diffusion (nav\_field conditioning, $K{=}20$)
\end{itemize}
\end{frame}

\begin{frame}{Phase B v1.0: What is ``Physics'' here? (Nav field prior)}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.52\linewidth}
    \begin{itemize}
      \item \textbf{Nav field} is estimated on the \textbf{train split only}: local mean-flow direction + count/speed statistics.
      \item At inference we crop a local patch around the last observed position and encode it as conditioning.
      \item Interpretation: a \textbf{local mean-field prior} that stabilizes directionality and improves best-of-$K$ coverage.
    \end{itemize}
  \end{column}
  \begin{column}{0.46\linewidth}
    \FitGraphic[height=0.62\textheight]{fig_geo_nav_field.pdf}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Phase B v1.0: Micro-Level Performance (coverage)}
\begin{itemize}
  \item Deterministic baseline has strong mean ADE/FDE (as expected for L2 regression).
  \item Diffusion/physics diffusion improve \textbf{best-of-$K$} (oracle) capability.
  \item Physics improves oracle upper bound over data-only diffusion.
\end{itemize}

\begin{frame}{Phase B v1.0: Why macro-loss ablation is insufficient}
\begin{itemize}
  \item Adding a macro loss on $x_0^{pred}$ is conceptually valid, but \textbf{noisy} at large timesteps ($t \approx T$).
  \item If not carefully gated/normalized, the model can satisfy macro loss with \textbf{high-frequency shortcuts} (jitter).
  \item Empirically: limited improvements (plateau region), motivating an architectural pivot.
\end{itemize}
\end{frame}

\section{Phase B v1.1: Residual Diffusion Fix}

\begin{frame}{Phase B v1.1 (link from v1.0 to fix)}
\textbf{Relation to Phase B v1.0:} v1.0 reveals a low-frequency bias (shrinkage) that is hard to correct via noisy macro proxies.

\vspace{0.6em}
\textbf{Phase B v1.1 objective:} recover macroscopic displacement \emph{without} sacrificing diversity and best-of-$K$ coverage.

\vspace{0.6em}
\textbf{Key idea:} let a deterministic model handle low-frequency scale; let diffusion model only stochastic residuals.
\end{frame}

\begin{frame}{Residual Diffusion (v1.1): Method}
\begin{block}{Decomposition}
\[
  \mathbf{v}_{1:F} = \mathbf{v}^{\mathrm{prior}}_{1:F} + \mathbf{v}^{\mathrm{res}}_{1:F},\qquad
  \mathbf{v}^{\mathrm{prior}}_{1:F} = f_{\mathrm{base}}(\mathbf{X}, \mathbf{c})
\]
\end{block}
\begin{itemize}
  \item Freeze $f_{\mathrm{base}}$ (baseline) as a deterministic prior (no gradient leakage).
  \item Train diffusion on residual targets: $\mathbf{v}^{\mathrm{res}} = \mathbf{v} - \mathbf{v}^{\mathrm{prior}}$.
  \item Inference: sample $\mathbf{v}^{\mathrm{res}}$ then add back $\mathbf{v}^{\mathrm{prior}}$.
\end{itemize}
\end{frame}

\begin{frame}{Residual Diffusion (v1.1): Framework Diagram}
\begin{figure}

\begin{frame}{Residual Diffusion (v1.1): Quick Validation}
\scriptsize
\begin{table}

\begin{itemize}
  \item Same conditions (val subset, 192 samples). Residual keeps strong best-of-$K$.
  \item Macro recovery: Rog/GT $\approx 0.93$, MSD$_{10}$/GT $\approx 0.84$, speed ratio $\approx 0.95$.
  \item Diversity check: ADE$_{\mathrm{std}}$ is non-zero (no residual collapse observed in quick eval).
\end{itemize}
\end{frame}

\begin{frame}{Residual Diffusion: What we learned}
\begin{itemize}
  \item The shrinkage is fundamentally a \textbf{low-frequency learning bias} (mean reversion).
  \item Residual decomposition is a \textbf{principled fix}: it separates ``scale'' (prior) from ``uncertainty'' (residual).
  \item Physics/nav\_field remains useful as a \textbf{directional guide} once scale is stabilized.
\end{itemize}
\end{frame}

\begin{frame}{Takeaways \& Next Steps}
\begin{itemize}
  \item Phase A: strict, leakage-free dt-fixed pipeline enables fair comparisons.
  \item Phase B v1.0: diffusion improves best-of-$K$ but fails macro realism (shrinkage).
  \item Phase B v1.1: residual diffusion recovers macro scale while preserving diversity.
  \item Next steps (paper-ready):
    \begin{itemize}
      \item Residual \textbf{physics diffusion} (nav\_field + residual prior).
      \item Strong generative baselines (SeqCVAE; recent diffusion predictors).
      \item Robustness: prior-swap test and additional geographic visualizations (zoomed regions / map tiles).
    \end{itemize}
\end{itemize}
\end{frame}
